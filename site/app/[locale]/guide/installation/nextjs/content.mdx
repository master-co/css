<StepSection>
    <Step $row>
        <StepL>
            ### <StepNum />Create a Next.js project
            If you don't have a Next.js project, create one first. It's recommended to refer to [Create Next App](https://nextjs.org/docs/app/api-reference/cli/create-next-app).
        </StepL>
        <StepR>
            ```bash name=Terminal
            npx create-next-app@latest my-app --ts --app --no-tailwind
            cd my-app
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Create a configuration file
            Run the command to create a [master.css](/guide/configuration) file.
        </StepL>
        <StepR>
            ```bash action=create
            @master/css@rc
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Install Master CSS
            Install Master CSS React and the Next.js adapter into your project via package managers.
        </StepL>
        <StepR>
            ```bash action=install
            @master/css.react@rc @master/css.next@rc
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Add client types
            Create a project-local environment file for the Master CSS config import types. Keep this separate from `next-env.d.ts`, because Next.js regenerates that file.
        </StepL>
        <StepR>
            ```ts name=master-css-env.d.ts
            /// <reference types="@master/css.next/client" />
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Set up the CSS runtime engine
            Create an App Router client boundary for `CSSRuntimeProvider` and load your `master.css` config through the Next.js adapter.

            (i) Keep the `?master-css-config` query. A plain `.css` import is treated as a stylesheet by Next.js, while this query loads the compiled Master CSS config object.
        </StepL>
        <StepR>
            ```tsx name=app/client.tsx
            'use client'

            import CSSRuntimeProvider from '@master/css.react'
            import config from '../master.css?master-css-config'

            export default function Client({ children }: { children: React.ReactNode }) {
                return (
                    <CSSRuntimeProvider config={config}>
                        {children}
                    </CSSRuntimeProvider>
                )
            }
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Author app CSS and config
            Keep the normal `app/globals.css` import in the root layout. The same stylesheet can hold regular CSS and `@master` definitions.

            (i) Import the stylesheet from `master.css` when you want its `@master {}` blocks to participate in the config graph. Ordinary CSS outside `@master` still follows the app stylesheet pipeline.
        </StepL>
        <StepR>
            ```css name=app/globals.css
            html,
            body {
                margin: 0;
                min-height: 100%;
            }

            .native-card {
                border: 1px solid rgb(0 0 0 / 12%);
            }

            @master {
                .hero-title {
                    @compose "m:0 font:5xl font:heavy fg:strong";
                }
            }
            ```

            ```css name=master.css
            @import './app/globals.css';
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Wrap the root layout
            Render the client boundary from `app/layout.tsx`. Progressive rendering hides development HTML until the runtime engine is ready.
        </StepL>
        <StepR>
            ```tsx name=app/layout.tsx
            import './globals.css'
            import Client from './client'

            export const metadata = {
                title: 'Create Next App',
                description: 'Generated by create next app',
            }

            export default function RootLayout({ children }: { children: React.ReactNode }) {
                return (
                    <html lang="en" hidden={process.env.NODE_ENV === 'development'}>
                        <body>
                            <Client>
                                {children}
                            </Client>
                        </body>
                    </html>
                )
            }
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Set up Next.js adapter
            Register the Master CSS Next.js adapter to pre-render CSS for build-time HTML outputs during `next build`.

            (i) This adapter uses the Next.js 16 Adapter API and runs in `onBuildComplete`. It pre-renders static and pre-rendered HTML outputs, while dynamic SSR responses continue to rely on the runtime engine.

            (i) Component CSS is generated when component classes are found in the built HTML or later observed by the runtime engine.
        </StepL>
        <StepR>
            ```js name=next.config.mjs
            import { withMasterCSS } from '@master/css.next'

            /** @type {import('next').NextConfig} */
            const nextConfig = withMasterCSS({
                /* your Next.js config */
            })

            export default nextConfig
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Launch server
            Run `npm run dev` to start your Next.js development server. Development uses the runtime engine directly for fast refresh.

            (i) Pre-rendered `<style id="master">` is generated by `next build`, so verify progressive output with `npm run build && npm run start`.
        </StepL>
        <StepR>
            ```bash name=Terminal
            npm run dev
            ```
        </StepR>
    </Step>
    <Step>
        ### <StepNum />Start using Master CSS
        Now style your first element using Master CSS syntax!
        ```tsx name=app/page.tsx
        export default function Home() {
            return (
                <h1 className="hero-title native-card m:2xl">Hello World</h1>
            )
        }
        ```
        Open your browser to watch the changes.
        <HelloWorld url="localhost:3000" />
    </Step>
</StepSection>
