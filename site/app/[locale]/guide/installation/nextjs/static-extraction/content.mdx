<StepSection>
    <Step $row>
        <StepL>
            ### <StepNum />Create a Next.js project
            If you don't have a Next.js project, create one first. It's recommended to refer to [Create Next App](https://nextjs.org/docs/app/api-reference/cli/create-next-app).
        </StepL>
        <StepR>
            ```bash name=Terminal
            npx create-next-app@latest my-app --ts --app --no-tailwind
            cd my-app
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Create a configuration file
            Run the command to create a [master.css](/guide/configuration) file.
        </StepL>
        <StepR>
            ```bash action=create
            @master/css@rc
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Install Master CSS
            Install the Master CSS extractor into your project via package managers.
        </StepL>
        <StepR>
            ```bash action=install
            @master/css.webpack@rc
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Set up CSS extractor
            Define one `MasterCSSExtractorPlugin` instance outside the exported Next.js config, then add it to Webpack.

            (!) Do not create `MasterCSSExtractorPlugin` inside the `webpack` callback. Next.js can call that callback more than once.
        </StepL>
        <StepR>
            ```ts name=next.config.ts
            import type { NextConfig } from 'next'
            import { MasterCSSExtractorPlugin } from '@master/css.webpack' // [!code ++]

            const masterCSS = new MasterCSSExtractorPlugin() // [!code ++]

            const nextConfig: NextConfig = {
                webpack(config) { // [!code ++]
                    config.plugins.push(masterCSS) // [!code ++]
                    return config // [!code ++]
                } // [!code ++]
            }

            export default nextConfig
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Import virtual CSS module
            Import the generated CSS module from `app/globals.css`.

            (i) `@import 'master.css'` and `@import 'virtual:master.css'` both point to the generated virtual CSS module in Webpack. Use one of them, not both.
        </StepL>
        <StepR>
            ```css name=app/globals.css
            @import 'master.css';
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Author app stylesheet config
            Source stylesheets can define `@master` classes and regular CSS. Regular class selectors are scanned like Tailwind-style native CSS and emitted only when used.

            (i) If a class is defined in both `app/globals.css` and the root `master.css`, the root `master.css` config is applied last.
        </StepL>
        <StepR>
            ```css name=app/globals.css
            @import 'master.css';

            .native-card {
                border: 1px solid rgb(0 0 0 / 12%);
            }

            .unused-card {
                color: red;
            }

            @master {
                .hero-title {
                    @compose "m:0 font:5xl font:heavy fg:strong";
                }
            }
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Launch server
            Run `npm run dev` to start your Next.js development server
        </StepL>
        <StepR>
            ```bash name=Terminal
            npm run dev
            ```
        </StepR>
    </Step>
    <Step>
        ### <StepNum />Start using Master CSS
        Now style your first element using Master CSS syntax!
        ```tsx name=app/page.tsx
        export default function Home() {
            return (
                <h1 className="hero-title native-card m:2xl">Hello World</h1>
            )
        }
        ```
        Open your browser to watch the changes.
        <HelloWorld url="localhost:3000" />
    </Step>
</StepSection>
