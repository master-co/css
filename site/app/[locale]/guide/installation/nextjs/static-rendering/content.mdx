    <StepR>
      ```bash name=Terminal
      npx create-next-app@latest my-app --ts --app --no-tailwind
      cd my-app
      ```
    </StepR>

  <Step $row>
    <StepL>
      ### <StepNum />Add Master CSS
      Run the installer in static mode from the Next.js project root. Static setup writes the generated stylesheet before Next.js loads the config.

      (i) Keep the installer-added `@import '@master/css';` in `app/globals.css`. Static mode replaces that import with the generated stylesheet and uses source scanning as the correctness baseline.
    </StepL>
    <StepR>
      ```bash name=Terminal
      npm create @master/css@rc -- --framework nextjs --mode static --yes
      ```
    </StepR>
  </Step>
  <Step $row>
    <StepL>
      ### <StepNum />Launch server
      Run the development server.
    </StepL>
    <StepR>
      ```bash name=Terminal&action=run
      dev
      ```
    </StepR>
  </Step>
  <Step>
    ### <StepNum />Start using Master CSS
    Now style your first page using Master CSS syntax.
  </Step>
