<StepSection>
    <Step $row>
        <StepL>
            ### <StepNum />Create a Lit project
            If you don't have a Lit project, create one first. It's recommended to refer to [Lit documentation](https://lit.dev/docs/).
        </StepL>
        <StepR>
            ```bash name=Terminal
            npm create vite@latest my-app -- --template lit-ts
            cd my-app
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Install Master CSS
            Install [`@master/css-runtime`](/reference/runtime) and [`@master/css.vite`](/guide/installation/vite) into your project.
        </StepL>
        <StepR>
            ```bash action=install
            @master/css-runtime@rc @master/css.vite@rc
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Set up Vite config
            Register the Master CSS Vite config loader without automatic runtime injection.
        </StepL>
        <StepR>
            ```ts name=vite.config.ts
            import { defineConfig } from 'vite'
            import masterCSS from '@master/css.vite' // [!code highlight]

            export default defineConfig({
                plugins: [
                    masterCSS({ mode: null }) // [!code highlight]
                ]
            })
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Import the default stylesheet
            Import Master CSS from the page stylesheet your Lit entry already loads. This stylesheet is also the project CSS entry.
        </StepL>
        <StepR>
            ```css name=src/index.css
            @import '@master/css';
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Add client types
            Add the Vite client types and Master CSS Vite client types before importing `virtual:master-css-config` from TypeScript.
        </StepL>
        <StepR>
            ```ts name=src/vite-env.d.ts
            /// <reference types="vite/client" />
            /// <reference types="@master/css.vite/client" />
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Set up CSS runtime engine
            Decorate your Lit element with `@cssRuntime()` to initialize Master CSS Runtime for the element's shadow root.
        </StepL>
        <StepR>
            ```ts name=src/my-element.ts
            import { LitElement, html } from 'lit'
            import { customElement } from 'lit/decorators.js'
            import { cssRuntime } from '@master/css-runtime' // [!code highlight]
            import type { CSSRuntime } from '@master/css-runtime' // [!code highlight]
            import config from 'virtual:master-css-config' // [!code highlight]

            @customElement('my-element')
            @cssRuntime({ config }) // [!code highlight]
            export class MyElement extends LitElement {
                cssRuntime?: CSSRuntime // [!code highlight]

                render() {
                    return html`
                        <h1 class="italic m:2xl fg:strong font:5xl font:heavy">
                            Hello World
                        </h1>
                    `
                }
            }
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Start dev server
            Run the command to start the Vite dev server.
        </StepL>
        <StepR>
            ```bash name=Terminal&action=run
            dev
            ```
        </StepR>
    </Step>
    <Step $row>
        <StepL>
            ### <StepNum />Start using Master CSS
            Now style elements inside your Lit templates using Master CSS syntax!
        </StepL>
        <StepR>
            ```ts name=src/my-element.ts
            render() {
                return html`
                    <button class="inline-flex items:center gap:xs px:md py:xs r:lg bg:blue fg:white">
                        Hello Lit
                    </button>
                `
            }
            ```
        </StepR>
    </Step>
    <HelloWorld url="localhost:5173" />
</StepSection>
