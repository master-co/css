The installer uses Rsbuild's `tools.rspack` hook to register the Webpack-compatible plugin and creates `src/index.css` as the Master CSS entry.

```ts name=rsbuild.config.ts
import { defineConfig } from '@rsbuild/core'
import MasterCSSPlugin from '@master/css-webpack'

export default defineConfig({
  tools: {
    rspack(config) {
      config.plugins ||= []
      config.plugins.push(new MasterCSSPlugin())
      return config
    }
  }
})
```

```css name=src/index.css
@import '@master/css';
```

```tsx
export default function App() {
  return (
    <h1 className="m:2xl italic font:5xl font:heavy text:neutral">
      Hello World
    </h1>
  )
}
```
